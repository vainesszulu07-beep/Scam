function addMessage(msg){ const div = document.createElement("div"); div.classList.add("message"); div.classList.add(msg.sender_id===userId ? "sent" : "received"); let contentHTML = ""; if(msg.content){ /* IMAGE */ if(msg.content.match(/\.(jpg|jpeg|png|gif|webp)$/i)){ contentHTML = ` `; } /* VIDEO */ else if(msg.content.match(/\.(mp4|webm|mov)$/i)){ contentHTML = ` `; } /* TEXT */ else{ contentHTML = msg.content; } } div.innerHTML = ` ${contentHTML} ${new Date(msg.created_at).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'})} `; messagesDiv.appendChild(div); messagesDiv.scrollTop = messagesDiv.scrollHeight; }
